module lut_dtc_top #(
    parameter N_TAP = 512,
    parameter CODE_WIDTH = 9
) (
    input  wire              sys_clk,
    input  wire              rst_n,
    input  wire              din,
    input  wire [CODE_WIDTH-1:0] code,
    input  wire              cal_mode,
    input  wire [CODE_WIDTH-1:0] cal_code,
    input  wire              dtc_sel,
    input  wire              vco_clk,
    output wire              dout,
    output wire  [CODE_WIDTH-1:0] cal_result,
    output wire               cal_valid
);

wire [4:0] div_out_dtc_i;
wire [19:0] cal_dtc_i;
wire [CODE_WIDTH-1:0] dtc_code;

assign div_out_dtc_i[0] = din;
// CARRY8 链输出
wire carry_dout;
assign carry_dout = (dtc_sel == 0) ? div_out_dtc_i[1] : div_out_dtc_i[4];
assign dtc_code = code;


// 输出选择
assign dout = carry_dout;

            (* DONT_TOUCH = "TRUE" *)
            lut_dtc #(
            .N_TAP (512),
            .CODE_WIDTH (9)
        ) u_lut_dtc_dly (
                     .sys_clk       (sys_clk),
                     .rst_n         (rst_n),
                     .din           (div_out_dtc_i[0]),
                     .code          (dtc_code),
                     .cal_mode      (1'b0),
                     .dtc_sel       (1'b0),
                     .dout          (div_out_dtc_i[1]),
                     .cal_result    (),
                     .cal_valid     ()
            );

            // 级 2~4 (3阶用, 4级链合计 ~8ns)

    genvar i;
    generate
        for (i = 1; i < 17; i = i + 1) begin : cascade

// ============================================================
    // TDC 延迟测量模块 (vco_clk domain, 无 CDC 偏差)
    // 测量 din 上升沿到 16 级 lut_dtc 延迟链输出上升沿的 vco_clk 周期数
    //
    // CDC fix: cal_dtc_i[0] 不再直接接 din(sys_clk域),
    // 而是接 din_d2 (2级同步到 vco_clk 域后的输出).
    // 这样校准链的起止都在同一时钟域, 消除 0~2 vco_clk 的测量偏差.
    // ============================================================
    localparam TDC_WIDTH   = 9;
     
    reg din_d1, din_d2;                // din 同步寄存器 (vco_clk 域)
    reg [TDC_WIDTH-1:0] tdc_counter;   // 计数器
    reg [TDC_WIDTH-1:0] tdc_count;     // 锁存结果
    reg tdc_measuring;                  // 测量中标志
    reg tdc_valid;                      // 结果有效

    // cal_mode CDC: sys_clk → vco_clk 域 (2-flop synchronizer)
    reg cal_mode_m1, cal_mode_m2;
    
    // 校准链输入取自同步后的 din_d2 (即 vco_clk 域)
    assign cal_dtc_i[0] = din_d2;
    
    // 16 级 lut_dtc 延迟后的输出 (组合逻辑, 但传播需 ~122ns)
    wire din_delayed = cal_dtc_i[16];

    // 边沿检测 (vco_clk 域, 无跨时钟域问题)
    wire din_rise = din_d1 && !din_d2;

    always @(posedge vco_clk or negedge rst_n) begin
        if (!rst_n) begin
            din_d1 <= 1'b0;
            din_d2 <= 1'b0;
            tdc_counter <= {TDC_WIDTH{1'b0}};
            tdc_measuring <= 1'b0;
            tdc_count <= {TDC_WIDTH{1'b0}};
            tdc_valid <= 1'b0;
            cal_mode_m1 <= 1'b0;
            cal_mode_m2 <= 1'b0;
        end else begin
            // 将 din (sys_clk 域) 同步到 vco_clk 域
            din_d1 <= din;
            din_d2 <= din_d1;

            // cal_mode CDC: sys_clk → vco_clk
            cal_mode_m1 <= cal_mode;
            cal_mode_m2 <= cal_mode_m1;

            // 测量逻辑 (使用同步后的 cal_mode_vco)
            if (cal_mode_m2 && tdc_valid) begin
                // 保持上次校准结果不变
                tdc_valid <= tdc_valid;
                tdc_count <= tdc_count;
            
            end else if (cal_mode_m2) begin
                if (din_rise) begin
                    // din 上升沿: 开始计数 (从1开始, 因为本条边沿不算)
                    tdc_counter <= {{(TDC_WIDTH-1){1'b0}}, 1'b1};
                    tdc_measuring <= 1'b1;
                    tdc_valid <= 1'b0;
                end else if (tdc_measuring) begin
                    // 直接采样延迟链输出 (均在 vco_clk 域, 无 CDC)
                    if (din_delayed) begin
                        tdc_count <= tdc_counter;
                        tdc_valid <= 1'b1;
                        tdc_measuring <= 1'b0;
                    end else begin
                        tdc_counter <= tdc_counter + 1'b1;
                        tdc_valid <= 1'b0;
                    end
                end else begin
                    tdc_valid <= 1'b0;
                end
            end else begin
                // 非测量模式
                tdc_measuring <= 1'b0;
                tdc_counter <= {TDC_WIDTH{1'b0}};
                tdc_valid <= 1'b0;
            end
        end
    end

    assign cal_result = tdc_count;
    assign cal_valid  = tdc_valid;
    

endmodule



===== src/dtc/lut_dtc.v =====
`timescale 1ns / 1ps
// 零延迟 DTC 模型（事件安全，适合仿真）
module lut_dtc #(parameter N_TAP=512, parameter CODE_WIDTH=9)(
    input wire sys_clk, input wire rst_n, input wire din,
    input wire [CODE_WIDTH-1:0] code, input wire cal_mode,
    input wire dtc_sel, output wire dout,
    output reg [CODE_WIDTH-1:0] cal_result, output reg cal_valid);
    assign dout = din;

    // 简化 TDC 模型
    reg [8:0] tdc_cnt = 0;
    reg [8:0] tdc_code = 0;
    reg tdc_busy = 0;

    always @(posedge sys_clk or negedge rst_n) begin
        if (!rst_n) begin
            cal_valid <= 1'b0; cal_result <= 0;
            tdc_cnt <= 0; tdc_code <= 0; tdc_busy <= 0;
        end else if (cal_mode && !tdc_busy) begin
            tdc_code <= code;
            tdc_cnt <= code / 10;
            tdc_busy <= 1;
            cal_valid <= 1'b0;
        end else if (tdc_busy) begin
            if (tdc_cnt == 0) begin
                tdc_busy <= 0;
                cal_valid <= 1'b1;
                cal_result <= tdc_code / 10;
            end else begin
                tdc_cnt <= tdc_cnt - 1;
                cal_valid <= 1'b0;
            end
        end else begin
            cal_valid <= 1'b0;
        end
    end
endmodule
